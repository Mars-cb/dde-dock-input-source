cmake_minimum_required(VERSION 3.13)
project(dde-dock-input-source VERSION 0.1.5 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(GNUInstallDirs)
include(CTest)
find_package(Qt5 5.11 REQUIRED COMPONENTS Core Widgets)
find_package(DtkWidget REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(DDCUTIL REQUIRED IMPORTED_TARGET ddcutil>=0.9.0)
pkg_check_modules(UDEV REQUIRED IMPORTED_TARGET libudev)
find_path(DOCK_INCLUDE_DIR pluginsiteminterface.h
    HINTS /usr/include/dde-dock "${CMAKE_CURRENT_SOURCE_DIR}/../dde-dock/interfaces"
    PATH_SUFFIXES dde-dock)
if(NOT DOCK_INCLUDE_DIR)
    message(FATAL_ERROR "Install dde-dock-dev or set DOCK_INCLUDE_DIR to the dock interface directory")
endif()
set(INPUT_SOURCE_PLUGIN_DIR "lib/dde-dock/plugins" CACHE STRING "DDE dock plugin install directory")
set(INPUT_SOURCE_HELPER_DIR "${CMAKE_INSTALL_LIBEXECDIR}/dde-dock-input-source")
add_library(input-source-model STATIC src/model.cpp)
target_include_directories(input-source-model PUBLIC src)
target_link_libraries(input-source-model PUBLIC Qt5::Core)
add_executable(dde-dock-input-source-helper src/helper.cpp)
target_link_libraries(dde-dock-input-source-helper PRIVATE input-source-model PkgConfig::DDCUTIL)
add_library(input-source-ui STATIC src/controller.cpp src/displaywatcher.cpp src/panel.cpp)
target_link_libraries(input-source-ui PUBLIC input-source-model Qt5::Widgets ${DtkWidget_LIBRARIES} PkgConfig::UDEV)
target_compile_definitions(input-source-ui PRIVATE
    INPUT_SOURCE_HELPER_PATH="${CMAKE_INSTALL_PREFIX}/${INPUT_SOURCE_HELPER_DIR}/dde-dock-input-source-helper")
add_library(dde-dock-input-source MODULE src/plugin.cpp)
# DDE filters the legacy libdde-dock-* filename prefix before loading.
set_target_properties(dde-dock-input-source PROPERTIES OUTPUT_NAME dock-input-source)
target_include_directories(dde-dock-input-source PRIVATE "${DOCK_INCLUDE_DIR}")
target_link_libraries(dde-dock-input-source PRIVATE input-source-ui)
add_executable(input-source-panel src/main.cpp)
target_link_libraries(input-source-panel PRIVATE input-source-ui)

install(TARGETS dde-dock-input-source LIBRARY DESTINATION "${INPUT_SOURCE_PLUGIN_DIR}")
install(TARGETS dde-dock-input-source-helper RUNTIME DESTINATION "${INPUT_SOURCE_HELPER_DIR}")
install(TARGETS input-source-panel RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
install(FILES data/60-dde-dock-input-source.rules DESTINATION "lib/udev/rules.d")
install(FILES README.md LICENSE DESTINATION "${CMAKE_INSTALL_DOCDIR}")
if(BUILD_TESTING)
    find_package(Qt5 REQUIRED COMPONENTS Test)
    add_executable(test-input-source tests/test_input_source.cpp)
    target_include_directories(test-input-source PRIVATE "${DOCK_INCLUDE_DIR}")
    target_link_libraries(test-input-source PRIVATE input-source-ui Qt5::Test)
    target_compile_definitions(test-input-source PRIVATE TEST_PLUGIN_PATH="$<TARGET_FILE:dde-dock-input-source>")
    add_dependencies(test-input-source dde-dock-input-source dde-dock-input-source-helper)
    add_test(NAME input-source COMMAND test-input-source)
    set_tests_properties(input-source PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
